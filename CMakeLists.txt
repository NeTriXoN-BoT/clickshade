cmake_minimum_required(VERSION 3.20)
project(ClickShade VERSION 1.0 LANGUAGES CXX)
option(CLICKSHADE_BUILD_TESTS "Build focused platform checks" OFF)
if(CLICKSHADE_BUILD_TESTS)
    enable_testing()
    add_executable(ClickShadeUiScaleTest tests/ui_scale_test.cpp)
    add_executable(ClickShadeFramePacingTest tests/frame_pacing_test.cpp)
    add_executable(ClickShadeShakeLocatorTest tests/shake_locator_test.cpp)
    foreach(target IN ITEMS ClickShadeUiScaleTest ClickShadeFramePacingTest ClickShadeShakeLocatorTest)
        target_include_directories(${target} PRIVATE src)
        target_compile_features(${target} PRIVATE cxx_std_17)
        if(MSVC)
            set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
            if(WIN32)
                target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
            endif()
        endif()
    endforeach()
    add_test(NAME UiScaleGeometry COMMAND ClickShadeUiScaleTest)
    add_test(NAME AdaptiveFramePacing COMMAND ClickShadeFramePacingTest)
    add_test(NAME ShakeLocator COMMAND ClickShadeShakeLocatorTest)
endif()
if(NOT WIN32)
    if(CLICKSHADE_BUILD_TESTS)
        enable_testing()
    endif()
    add_subdirectory(linux)
    return()
endif()
enable_language(RC)
add_executable(ClickShade WIN32
    src/app.cpp src/ui_layout.cpp src/ui_paint.cpp src/ui_actions.cpp src/ui_sounds.cpp
    src/runtime.cpp src/cursor_engine.cpp src/settings.cpp src/recovery.cpp src/crash_guard.cpp
    src/visual_effects.cpp src/click_audio.cpp
    resources/app.rc)
add_custom_command(TARGET ClickShade POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/sounds" "$<TARGET_FILE_DIR:ClickShade>/sounds"
    VERBATIM)
target_compile_features(ClickShade PRIVATE cxx_std_17)
target_compile_definitions(ClickShade PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN OEMRESOURCE _WIN32_WINNT=0x0A00)
target_include_directories(ClickShade PRIVATE resources)
target_link_libraries(ClickShade PRIVATE user32 gdi32 comctl32 comdlg32 shell32 dwmapi advapi32 wtsapi32 uxtheme ole32 oleaut32 secur32 winmm mfuuid)
if(MSVC)
    target_compile_options(ClickShade PRIVATE /W4 /permissive- /utf-8 /EHsc)
    set_property(TARGET ClickShade PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_options(ClickShade PRIVATE /MANIFEST:NO)
else()
    target_compile_options(ClickShade PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(ClickShade PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
